function P10Norm($dir) {   # temporary normalised hash, snapshot only
  $f = Join-Path $dir 'src\core\content.gd'
  $t = [IO.File]::ReadAllText($f)
  $old = "`t`th.update(FileAccess.get_file_as_bytes(f))"
  $new = "`t`tvar p10raw := FileAccess.get_file_as_bytes(f)`n`t`tvar p10out := PackedByteArray()`n`t`tfor p10c in p10raw:`n`t`t`tif p10c != 13: p10out.append(p10c)`n`t`th.update(p10out)"
  if (-not $t.Contains($old)) { throw "content.gd hash line not found in $dir" }
  [IO.File]::WriteAllText($f, $t.Replace($old, $new), $p10U8)
}
function P10Lf($dir) {      # replace data/ with an autocrlf=false export of the same commit
  Remove-Item -Recurse -Force (Join-Path $dir 'data')
  Push-Location $p10Root; git -c core.autocrlf=false archive --format=zip -o "$dir-d.zip" HEAD data; Pop-Location
  Expand-Archive "$dir-d.zip" -DestinationPath $dir; Remove-Item "$dir-d.zip"
}
function P10Edit($dir, $rel, $old, $new) {   # byte-level text edit, keeps the line endings of the file
  $f = Join-Path $dir $rel; $t = [IO.File]::ReadAllText($f)
  if (-not $t.Contains($old)) { throw "edit target not found in $rel" }
  [IO.File]::WriteAllText($f, $t.Replace($old, $new), $p10U8)
}

function P10Step($label, $proj, $mode) {
  $r = & $p10Wrap -GodotArgs @('--headless', '--path', $proj, '-s', 'res://docs/research/systems/probes/probe_update_scenario.gd', '--quit-after', '600', '--', "--mode=$mode") -WorkDir $p10Iso -Keep -TimeoutSec 120
  $ud = Join-Path $p10Iso 'appdata\Godot\app_userdata\Lane Defence'
  $hm = if (Test-Path "$ud\save.json") { (Get-FileHash "$ud\save.json").Hash.Substring(0, 12) } else { '' }
  $hb = if (Test-Path "$ud\save.json.bak") { (Get-FileHash "$ud\save.json.bak").Hash.Substring(0, 12) } else { '' }
  $files = (Get-ChildItem $ud -File -Filter 'save*' | ForEach-Object Name) -join ','
  "STEP $label main_sha=$hm bak_sha=$hb files=$files " + (($r | Where-Object { $_ -like 'PROBE10_UPDATE *' }) -replace '^PROBE10_UPDATE ', '')
  ($r | Where-Object { $_ -like 'ISOLATED_GODOT*' })
  $r | Where-Object { $_ -match 'SCRIPT ERROR|Parse Error|refund_unknown' }
}
