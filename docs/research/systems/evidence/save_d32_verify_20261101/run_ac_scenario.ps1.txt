function P10Lf($dir) {
  Remove-Item -Recurse -Force (Join-Path $dir 'data')
  Push-Location $p10Root; git -c core.autocrlf=false archive --format=zip -o "$dir-d.zip" HEAD data; Pop-Location
  Expand-Archive "$dir-d.zip" -DestinationPath $dir; Remove-Item "$dir-d.zip"
}
function P10Edit($dir, $rel, $old, $new) {
  $p10f = Join-Path $dir $rel; $p10t = [IO.File]::ReadAllText($p10f)
  if (-not $p10t.Contains($old)) { throw "edit target not found in $rel" }
  [IO.File]::WriteAllText($p10f, $p10t.Replace($old, $new), $p10U8)
}
