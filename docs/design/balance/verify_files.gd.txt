extends SceneTree
## 01-director post-edit verification on the REAL data files (no in-memory changes).
## Checks per round: content warnings; 02 competent bot (skill 2) seeds 11/22/33; 02 casual bot (skill 1) seed 9;
## passive player (no input) seed 5 must LOSE BY BASE DESTRUCTION (player base hp <= 0, t < time limit),
## a timeout does not count. Run: copy to $env:TEMP\verify_files.gd, then
##   tools\engine\Godot_v4.7.2-stable_win64_console.exe --headless --path . -s $env:TEMP\verify_files.gd -- <out.json> [r01,r02,...]
const ContentS := preload("res://src/core/content.gd")
const SimS := preload("res://src/gameplay/battle_sim.gd")
const BotS := preload("res://src/gameplay/bot_player.gd")

func _run(sim, bot) -> void:
	var step := 1.0 / 60.0
	while not sim.ended and sim.t < 900.0:
		if bot.skill > 0: bot.act(sim, step)
		sim.step(step)

func _initialize() -> void:
	var args := OS.get_cmdline_user_args()
	var out_path: String = args[0] if args.size() > 0 else "user://verify_files.json"
	var content = ContentS.new()
	content.load_all("res://data")
	var only: Array = []
	if args.size() > 1: only = Array(args[1].split(","))
	var rows := []
	var passive_ok := true
	for rid in content.round_order:
		if not only.is_empty() and not only.has(rid): continue
		var comp := []
		for sd in [11, 22, 33]:
			var s = SimS.new(); s.setup(content, rid, sd); _run(s, BotS.new(2, sd))
			comp.append({"seed": sd, "won": s.won, "t": snappedf(s.t, 0.1),
				"pbase": snappedf(float(s.player_base.hp) / float(s.player_base.hp_max), 0.01), "spawned": s.stats.enemy_spawned})
		var c = SimS.new(); c.setup(content, rid, 9); _run(c, BotS.new(1, 9))
		var p = SimS.new(); p.setup(content, rid, 5); _run(p, BotS.new(0, 5))
		var destroyed: bool = float(p.player_base.hp) <= 0.0 and p.t < 899.0
		if p.won or not destroyed: passive_ok = false
		rows.append({"round": rid, "ai_pool": content.rounds[rid].enemy_ai.get("pool", []), "competent": comp,
			"casual_won": c.won, "casual_t": snappedf(c.t, 0.1),
			"passive_won": p.won, "passive_t": snappedf(p.t, 0.1), "passive_base_destroyed": destroyed})
	var res := {"content_warnings": Array(content.warnings), "passive_all_lose_by_destruction": passive_ok, "rows": rows}
	var f := FileAccess.open(out_path, FileAccess.WRITE)
	f.store_string(JSON.stringify(res, " ")); f.close()
	print("VERIFY_DONE passive_all_lose_by_destruction=", passive_ok, " warnings=", content.warnings.size())
	var ok: bool = passive_ok and content.warnings.size() == 0
	content.free()
	quit(0 if ok else 1)
