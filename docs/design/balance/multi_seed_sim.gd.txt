extends SceneTree
## 01-director multi-seed balance check on the REAL data (02's unmodified battle_sim.gd + competent bot_player.gd).
## With the random enemy AI off, the sim itself is deterministic, so different sim seeds alone give identical runs.
## Variance therefore comes from PLAYER TIMING NOISE, seeded per run: the bot first acts after U(0,3) s, and after each
## decision waits an extra U(0,0.35) s (models human reaction time). This is a stand-in for human variance, not a human model.
## Extra probe for r01 risk: a "late starter" bot that does nothing for the first N s (arg 6, default 30).
## Args: <out.json> [overrides.json|-] [budgets=none,low,high] [seeds=1,2,3,4,5] [rounds=all|r01,r05] [late_delay_s=30] [policies=competent,casual]
## Policy "casual" = 02 bot_player skill 1 (the same casual policy 07 uses in probe_balance.gd).
const ContentS := preload("res://src/core/content.gd")
const SimS := preload("res://src/gameplay/battle_sim.gd")
const BotS := preload("res://src/gameplay/bot_player.gd")

func _run(sim, bot, seed_v: int, start_delay: float) -> Dictionary:
	var step := 1.0 / 60.0
	var rng := RandomNumberGenerator.new(); rng.seed = seed_v * 7919 + 13
	var wait := maxf(start_delay, rng.randf_range(0.0, 3.0))
	var base_dmg := 0.0
	var first_hit := -1.0
	var w1_clear := -1.0
	var ev_seen := 0
	var prev: float = float(sim.player_base.hp)
	var since_act := 0.0
	var boss_in := -1.0
	var hit_src := {}          # D36: base damage split by source ("wave<N>" scripted wave, "other" = trickle/trigger/pressure)
	var first_hit_who := []    # enemy ids + wave tag near the base at the first base hit
	var boss_out := -1.0
	while not sim.ended and sim.t < 900.0:
		if bot != null:
			wait -= step
			since_act += step
			if wait <= 0.0:
				# FIX 2026-10-31 (D30): pass the real elapsed time. Before this fix the bot got dt=1/60 per call although
				# calls were ~0.17 s apart, so its internal think timer ran ~10x slow (a slowed-down bot).
				bot.act(sim, since_act)
				since_act = 0.0
				wait = rng.randf_range(0.0, 0.35)
		sim.step(step)
		var hp: float = float(sim.player_base.hp)
		if hp < prev:
			base_dmg += prev - hp
			var near := []
			for e in sim.enemies:
				if not e.dying and absf(e.x - float(sim.player_base.x)) <= float(e.def.range) + 40.0:
					near.append(e)
			for e in near:
				var key: String = ("wave%d" % e.wave_tag) if e.wave_tag > 0 else "other"
				hit_src[key] = float(hit_src.get(key, 0.0)) + (prev - hp) / float(near.size())
			if first_hit < 0.0:
				first_hit = sim.t
				for e in near: first_hit_who.append("%s@%s" % [str(e.def.id), ("w%d" % e.wave_tag) if e.wave_tag > 0 else "other"])
				first_hit_who.append("enemy_base=%.2f" % (float(sim.enemy_base.hp) / float(sim.enemy_base.hp_max)))
		prev = hp
		var boss_alive := false
		for e in sim.enemies:
			if e.is_boss and not e.dying: boss_alive = true
		if boss_alive and boss_in < 0.0: boss_in = sim.t
		if boss_in >= 0.0 and boss_out < 0.0 and not boss_alive: boss_out = sim.t
		while ev_seen < sim.events.size():
			var ev = sim.events[ev_seen]; ev_seen += 1
			if ev[0] == "wave_cleared" and int(ev[1].get("index", 0)) == 1 and w1_clear < 0.0: w1_clear = float(ev[1].get("t", sim.t))
		if ev_seen > 4000:
			sim.events.clear(); ev_seen = 0
	return {"won": sim.won, "t": snappedf(sim.t, 0.1), "pbase": snappedf(float(sim.player_base.hp) / float(sim.player_base.hp_max), 0.01),
		"base_dmg": snappedf(base_dmg, 1.0), "first_base_hit_t": snappedf(first_hit, 0.1), "wave1_cleared_t": snappedf(w1_clear, 0.1), "boss_spawn_t": snappedf(boss_in, 0.1), "boss_dead_t": snappedf(boss_out, 0.1), "hit_src": hit_src, "first_hit_who": first_hit_who, "stars": int(sim.result().stars) if sim.won else 0}

func _initialize() -> void:
	var a := OS.get_cmdline_user_args()
	var out_path: String = a[0] if a.size() > 0 else "user://multi_seed.json"
	var overrides := {}
	if a.size() > 1 and a[1] != "-" and FileAccess.file_exists(a[1]):
		overrides = JSON.parse_string(FileAccess.get_file_as_string(a[1]))
	var budgets_list: Array = Array((a[2] if a.size() > 2 else "none,low,high").split(","))
	var seeds: Array = []
	for s in (a[3] if a.size() > 3 else "1,2,3,4,5").split(","): seeds.append(int(s))
	var only: Array = [] if a.size() <= 4 or a[4] == "all" else Array(a[4].split(","))
	var late_delay: float = float(a[5]) if a.size() > 5 else 30.0
	var policies: Array = Array((a[6] if a.size() > 6 else "competent").split(","))
	var budgets = JSON.parse_string(FileAccess.get_file_as_string("res://docs/design/balance/upgrade_budgets.json"))
	var content = ContentS.new()
	content.load_all("res://data")
	if overrides.has("_units"):
		for uid in overrides["_units"]:
			for k in overrides["_units"][uid]: content.units[uid][k] = overrides["_units"][uid][k]
	if overrides.has("_economy"):
		for k in overrides["_economy"]: content.economy[k] = overrides["_economy"][k]
	if overrides.has("_per_level"):
		for u in content.meta.get("upgrades", []):
			if overrides["_per_level"].has(str(u.id)): u["per_level"] = overrides["_per_level"][str(u.id)]
	var rows := []
	for rid in content.round_order:
		if not only.is_empty() and not only.has(rid): continue
		if overrides.has(rid):
			for k in overrides[rid]: content.rounds[rid][k] = overrides[rid][k]
		for b in budgets_list:
			var lv: Dictionary = {} if b == "none" else budgets[b][rid]["levels"]
			for pol in policies:
				for sd in seeds:
					var s = SimS.new(); s.setup(content, rid, sd, lv)
					var r := _run(s, BotS.new(2 if pol == "competent" else 1, sd), sd, 0.0)
					r.merge({"round": rid, "budget": b, "seed": sd, "kind": pol})
					rows.append(r)
		var p = SimS.new(); p.setup(content, rid, 5)
		var pr := _run(p, null, 5, 0.0); pr.merge({"round": rid, "budget": "none", "seed": 5, "kind": "idle"}); rows.append(pr)
		if rid == "r01":
			for sd in seeds:
				var l = SimS.new(); l.setup(content, rid, sd)
				var lr := _run(l, BotS.new(2, sd), sd, late_delay); lr.merge({"round": rid, "budget": "none", "seed": sd, "kind": "late%d" % int(late_delay)}); rows.append(lr)
	var f := FileAccess.open(out_path, FileAccess.WRITE)
	f.store_string(JSON.stringify({"warnings": Array(content.warnings), "seeds": seeds, "rows": rows}, " ")); f.close()
	print("MULTI_SEED_DONE rows=", rows.size())
	content.free()
	quit(0)
